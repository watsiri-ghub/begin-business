การเชื่อมต่อคลาวด์ Google Sheets

วิธีการเชื่อมโยงข้อมูลแบบ 2 ทาง:

  1. เปิด Google Sheets เลือก Extensions > Apps Script
  2. ลบโค้ดเดิมทั้งหมด แล้ววางโค้ดด้านล่างลงไปแทน กด Save (รูปแผ่นดิสค์)
  3. กด Deploy > New deployment (ตั้งค่า Who has access เป็น Anyone) นำลิงก์ Web App URL มาวางช่องด้านล่าง
// ระบบจะสร้างแท็บ Transactions, Contracts, Categories ให้อัตโนมัติในสเปรดชีตนี้

function getOrCreateSheet(name, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  if (sheet.getLastRow() === 0) sheet.appendRow(headers);
  return sheet;
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function findRowById(sheet, id) {
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(id)) return i + 1;
  }
  return -1;
}

function formatDateCell(cell) {
  if (!cell) return '';
  if (Object.prototype.toString.call(cell) === '[object Date]') {
    return Utilities.formatDate(cell, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(cell).slice(0, 10);
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) return jsonOut({ result: "error", message: "no data" });
    var data = JSON.parse(e.postData.contents);
    var entity = data.entity || 'transaction';
    var action = data.action || 'add';

    if (entity === 'contract') return handleContract(data, action);
    if (entity === 'category') return handleCategory(data, action);
    return handleTransaction(data, action);
  } catch (err) {
    return jsonOut({ result: "error", message: err.toString() });
  }
}

function handleTransaction(data, action) {
  var sheet = getOrCreateSheet('Transactions', ["ID", "วันที่", "ประเภท", "หมวดหมู่", "เลขสัญญา", "จำนวนเงิน", "ช่องทาง", "หมายเหตุ"]);
  if (action === 'syncAll' && Array.isArray(data.transactions)) {
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) sheet.deleteRows(2, lastRow - 1);
    data.transactions.forEach(function(t) {
      sheet.appendRow([t.id, t.date, t.type, t.categoryName, t.contractCode || "-", t.amount, t.method, t.note]);
    });
    return jsonOut({ result: "success" });
  }
  var rowIndex = findRowById(sheet, data.id);
  if (action === 'delete') {
    if (rowIndex > -1) sheet.deleteRow(rowIndex);
    return jsonOut({ result: "success" });
  }
  var rowData = [data.id, data.date, data.type, data.categoryName, data.contractCode || "-", data.amount, data.method, data.note];
  if (rowIndex > -1) sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]);
  else sheet.appendRow(rowData);
  return jsonOut({ result: "success" });
}

  }
  var rowIndex = findRowById(sheet, data.id);

  else sheet.appendRow(rowData);
  return jsonOut({ result: "success" });
}

function doGet(e) {
  try {
    return jsonOut({
      status: "success",
      transactions: readTransactions(),
      contracts: readContracts(),
      categories: readCategories()
    });
  } catch (err) {
    return jsonOut({ status: "error", message: err.toString() });
  }
}

function readTransactions() {
  var sheet = getOrCreateSheet('Transactions', ["ID", "วันที่", "ประเภท", "หมวดหมู่", "เลขสัญญา", "จำนวนเงิน", "ช่องทาง", "หมายเหตุ"]);
  var rows = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0]) continue;
    out.push({
      id: String(r[0]),
      date: formatDateCell(r[1]) || new Date().toISOString().slice(0, 10),
      type: r[2] === 'รายรับ' ? 'income' : 'expense',
      categoryName: r[3],
      contractCode: r[4],
      amount: Number(r[5]) || 0,
      method: r[6] || 'ไม่ระบุ',
      note: r[7] || ''
    });
  }
  return out;
}

function readContracts() {
  var sheet = getOrCreateSheet('Contracts', ["ID", "เลขที่สัญญา", "ชื่อโครงการ", "มูลค่าสัญญา", "สถานะ", "วันลงนาม", "วันเริ่ม", "วันสิ้นสุด", "หมายเหตุ"]);
  var rows = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0]) continue;
    out.push({
      id: String(r[0]),
      code: r[1] || '',
      title: r[2] || '',
      amount: Number(r[3]) || 0,
      status: r[4] || 'active',
      signDate: formatDateCell(r[5]),
      startDate: formatDateCell(r[6]),
      endDate: formatDateCell(r[7]),
      note: r[8] || ''
    });
  }
  return out;
}

function readCategories() {
  var sheet = getOrCreateSheet('Categories', ["ID", "ชื่อหมวดหมู่", "ประเภท", "ไอคอน", "สี"]);
  var rows = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0]) continue;
    out.push({
      id: String(r[0]),
      name: r[1] || '',
      type: r[2] === 'income' ? 'income' : 'expense',
      icon: r[3] || 'tag',
      color: r[4] || '#64748b'
    });
  }
  return out;
}
